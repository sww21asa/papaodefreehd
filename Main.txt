local function sendCommand(cmd)
	if cmd == "" then return end
	pcall(function()
		local remote = ReplicatedStorage:FindFirstChild("HDAdminClient")
		if remote then
			local signals = remote:FindFirstChild("Signals")
			if signals then
				local request = signals:FindFirstChild("RequestCommand")
				if request then
					request:InvokeServer(";" .. cmd .. " ")
				end
			end
		end
	end)
end

	local isSpamming = false
	local connection = nil
	
	local function updateButton()
		spamBtn.Image = isSpamming and "rbxassetid://15145875494" or "rbxassetid://15145875494"
	end
	
	local function startSpam()
		if isSpamming then return end
		local delay = tonumber(DelayBox.Text) or 0.5
		delay = math.max(0.1, delay)
		isSpamming = true
		updateButton()
		
		connection = RunService.Heartbeat:Connect(function()
			if isSpamming and cmdBox.Text ~= "" then
				sendCommand(cmdBox.Text)
				task.wait(delay)
			end
		end)
	end
	
	local function stopSpam()
		if not isSpamming then return end
		isSpamming = false
		if connection then
			connection:Disconnect()
			connection = nil
		end
		updateButton()
	end
	
	spamBtn.MouseButton1Click:Connect(function()
		if isSpamming then stopSpam() else startSpam() end
	end)
	
	removeBtn.MouseButton1Click:Connect(function()
		stopSpam()
		container:Destroy()
	end)
	
	spamBtn.MouseEnter:Connect(function()
		TweenService:Create(spamBtn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(100, 0, 0)}):Play()
	end)
	spamBtn.MouseLeave:Connect(function()
		TweenService:Create(spamBtn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(139, 0, 0)}):Play()
	end)
	
	removeBtn.MouseEnter:Connect(function()
		TweenService:Create(removeBtn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(30, 30, 30)}):Play()
		removeBtn.TextColor3 = Color3.fromRGB(100, 0, 0)
	end)
	removeBtn.MouseLeave:Connect(function()
		TweenService:Create(removeBtn, TweenInfo.new(0.15), {BackgroundColor3 = Color3.fromRGB(0, 0, 0)}):Play()
		removeBtn.TextColor3 = Color3.fromRGB(200, 30, 30)
	end)
	
	return {container = container, cmdBox = cmdBox, stopSpam = stopSpam}
end
